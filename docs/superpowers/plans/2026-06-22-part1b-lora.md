- [ ] Step 2: Test `test_loralinear_scaling`: forzando A=ones, B=ones, output == `(alpha/rank) * rank * x_sum`-coerente; verifica che raddoppiando alpha l'output raddoppia.
- [ ] Step 3: Implementa `LoRALinear`:

```python
class LoRALinear(nn.Module):
    """Adapter LoRA: ΔW = (alpha/rank) * B @ A, con A random e B=0 a init.

    A init a 0 il delta è nullo -> il modello parte identico al pre-addestrato.
    Solo A e B sono parametri (i pesi base restano altrove, congelati).
    """
    def __init__(self, in_features, out_features, rank, alpha):
        super().__init__()
        self.rank = rank
        self.scaling = alpha / rank
        self.lora_A = nn.Linear(in_features, rank, bias=False)
        self.lora_B = nn.Linear(rank, out_features, bias=False)
        nn.init.normal_(self.lora_A.weight, mean=0.0, std=0.02)
        nn.init.zeros_(self.lora_B.weight)  # B=0 -> delta nullo a init

    def forward(self, x):
        return self.lora_B(self.lora_A(x)) * self.scaling
```

class GPT2_LoRA(GPT2LMHeadModel):
    def __init__(self, *model_args, rank, alpha, **model_kwargs):
        super().__init__(*model_args, **model_kwargs)
        for block in self.transformer.h:
            new_attn = CustomGPT2Attention(self.config, rank=rank, alpha=alpha)
            # copia i pesi base nel nuovo modulo (i lora_* restano a init);
            # i nomi dei parametri restano identici -> from_pretrained li popolerà
            new_attn.load_state_dict(block.attn.state_dict(), strict=False)
            block.attn = new_attn

    def forward(self, *args, **kwargs):
        return super().forward(*args, **kwargs)
```

def apply_lora_freezing(model):
    """Congela tutto tranne gli adapter LoRA."""
    for p in model.parameters():
        p.requires_grad = False
    for name, p in model.named_parameters():
        if "lora_" in name:
            p.requires_grad = True
    return model

def param_stats(model):
    total = sum(p.numel() for p in model.parameters())
    trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
    print(f"total params: {total:,}")
    print(f"trainable params: {trainable:,}")
    print(f"frozen params: {total - trainable:,}")
    return total, trainable
```
